import 'package:flutter/material.dart';

import '../../features/dragonball/presentation/app_settings.dart';
import 'app_tokens.dart';

/// LEARN (fonts): declare this family in pubspec.yaml (see INTEGRATION.md).
/// If the font files are missing, Flutter silently uses the default font,
/// so nothing breaks while you are still downloading them.
const String kHeadingFont = 'Rajdhani';

// LEARN (theming): ThemeData is built ONCE here, not styled widget by widget.
// Steps: 1) ColorScheme  2) TextTheme  3) per-component themes.
// Widgets then read Theme.of(context) and automatically match.
ThemeData buildAppTheme(AppSettings settings, Brightness brightness) {
  final isLight = brightness == Brightness.light;

  // 1) Colors. The seed (orange by default) generates a full Material 3
  // scheme; then we override two roles by hand so the app has a deliberate
  // second identity color (Saiyan blue) and an accent (aura gold).
  final generated = ColorScheme.fromSeed(
    seedColor: settings.seedColor,
    brightness: brightness,
    contrastLevel: settings.highContrast ? 0.7 : 0.0,
  );
  final scheme = generated.copyWith(
    secondary: isLight ? const Color(0xFF1F6FEB) : const Color(0xFF6CA8FF),
    tertiary: isLight ? const Color(0xFFB7791F) : const Color(0xFFFFC94D),
  );

  final background =
      isLight ? const Color(0xFFFFFAF5) : const Color(0xFF14161C);

  final base = ThemeData(
    useMaterial3: true,
    brightness: brightness,
    colorScheme: scheme,
  );

  // 2) Typography: only the big headings use the display font.
  TextStyle heading(TextStyle? style) => (style ?? const TextStyle()).copyWith(
        fontFamily: kHeadingFont,
        fontWeight: FontWeight.w700,
        letterSpacing: 0.2,
      );
  final text = base.textTheme;

  final cardRadius = settings.compactCards ? AppRadius.md : AppRadius.lg;
  final borderColor =
      settings.highContrast ? scheme.outline : scheme.outlineVariant;

  OutlineInputBorder inputBorder(Color color, [double width = 1]) =>
      OutlineInputBorder(
        borderRadius: AppRadius.mdAll,
        borderSide: BorderSide(color: color, width: width),
      );

  // 3) Component themes.
  return base.copyWith(
    scaffoldBackgroundColor: background,
    visualDensity:
        settings.compactCards ? VisualDensity.compact : VisualDensity.standard,
    textTheme: text.copyWith(
      headlineLarge: heading(text.headlineLarge),
      headlineMedium: heading(text.headlineMedium),
      headlineSmall: heading(text.headlineSmall),
      titleLarge: heading(text.titleLarge),
    ),
    appBarTheme: AppBarTheme(
      centerTitle: false,
      backgroundColor: background,
      foregroundColor: scheme.onSurface,
      surfaceTintColor: Colors.transparent,
      scrolledUnderElevation: 0,
    ),
    cardTheme: CardThemeData(
      elevation: 0,
      margin: EdgeInsets.zero,
      clipBehavior: Clip.antiAlias,
      color: scheme.surfaceContainerLow,
      shape: RoundedRectangleBorder(
        borderRadius: BorderRadius.circular(cardRadius),
        // A subtle border gives cards hierarchy on the flat background.
        side: BorderSide(color: borderColor),
      ),
    ),
    dividerTheme: DividerThemeData(
      color: borderColor,
      thickness: settings.highContrast ? 1.2 : 0.7,
    ),
    chipTheme: ChipThemeData(
      shape: const StadiumBorder(),
      side: BorderSide(color: borderColor),
      showCheckmark: false,
      selectedColor: scheme.primary.withValues(alpha: 0.18),
    ),
    navigationBarTheme: NavigationBarThemeData(
      backgroundColor: scheme.surfaceContainerLow,
      surfaceTintColor: Colors.transparent,
      indicatorColor: scheme.primary.withValues(alpha: 0.18),
    ),
    inputDecorationTheme: InputDecorationTheme(
      filled: true,
      fillColor: isLight ? Colors.white : scheme.surfaceContainerHigh,
      contentPadding: const EdgeInsets.symmetric(horizontal: 16, vertical: 14),
      border: inputBorder(Colors.transparent),
      enabledBorder: inputBorder(borderColor),
      focusedBorder: inputBorder(scheme.primary, 2),
    ),
    snackBarTheme: SnackBarThemeData(
      behavior: SnackBarBehavior.floating,
      backgroundColor: scheme.inverseSurface,
      contentTextStyle: TextStyle(color: scheme.onInverseSurface),
      shape: RoundedRectangleBorder(borderRadius: AppRadius.mdAll),
    ),
    bottomSheetTheme: BottomSheetThemeData(
      showDragHandle: true,
      backgroundColor: scheme.surfaceContainerLow,
      shape: const RoundedRectangleBorder(
        borderRadius:
            BorderRadius.vertical(top: Radius.circular(AppRadius.lg)),
      ),
    ),
  );
}

  static const double x4 = 16;
  static const double x5 = 20;
  static const double x6 = 24;

  static const double sm = 8;
  static const double md = 12;
  static const double lg = 20;

  static BorderRadius get mdAll => BorderRadius.circular(md);
  static BorderRadius get lgAll => BorderRadius.circular(lg);

/// Animation durations + one shared curve, so motion feels consistent.
abstract final class AppMotion {
  static const Duration fast = Duration(milliseconds: 150);
  static const Duration base = Duration(milliseconds: 250);
  static const Duration slow = Duration(milliseconds: 450);
  static const Curve curve = Curves.easeOutCubic;
}

/// LEARN (responsive layout): width -> number of grid columns, plus a
/// maximum content width so web / tablet layouts do not stretch forever.
abstract final class Breakpoints {
  static const double medium = 600;
  static const double large = 900;
  static const double xl = 1200;
  static const double maxContent = 1100;

  static int gridColumns(double width) {
    if (width < medium) return 2;
    if (width < large) return 3;
    if (width < xl) return 4;
    return 5;
  }

import 'package:flutter/foundation.dart';
import 'package:flutter_riverpod/flutter_riverpod.dart';
import 'package:shared_preferences/shared_preferences.dart';

import 'app_strings.dart';

// LEARN (Riverpod):
//  * languageProvider  -> state that can change (a Notifier)
//  * stringsProvider   -> DERIVED state: rebuilt automatically whenever the
//    language changes. Widgets just `ref.watch(stringsProvider)`.
//  * `.select(...)` -> watch only ONE field, so unrelated changes do not
//    rebuild the derived provider.

@immutable
class LanguageSettings {
  const LanguageSettings({
    this.app = AppLanguage.en,
    this.content = ContentLanguage.es,
  });

  final AppLanguage app;
  final ContentLanguage content;

  LanguageSettings copyWith({AppLanguage? app, ContentLanguage? content}) =>
      LanguageSettings(app: app ?? this.app, content: content ?? this.content);
}

final languageProvider =
    NotifierProvider<LanguageController, LanguageSettings>(
  LanguageController.new,
);

final stringsProvider = Provider<AppStrings>(
  (ref) => AppStrings(ref.watch(languageProvider.select((s) => s.app))),
);

class LanguageController extends Notifier<LanguageSettings> {
  static const _appKey = 'lang_app';
  static const _contentKey = 'lang_content';

  bool _changedByUser = false;

  @override
  LanguageSettings build() {
    _restore(); // async; the UI starts with defaults, then updates.
    return const LanguageSettings();
  }

  T _byName<T extends Enum>(List<T> values, String? name, T fallback) =>
      values.where((v) => v.name == name).firstOrNull ?? fallback;

  Future<void> _restore() async {
    try {
      final prefs = await SharedPreferences.getInstance();
      // Same guard as AppSettings: do not overwrite a choice the user made
      // while storage was still loading.
      if (_changedByUser) return;

      state = LanguageSettings(
        app: _byName(AppLanguage.values, prefs.getString(_appKey), AppLanguage.en),
        content: _byName(
          ContentLanguage.values,
          prefs.getString(_contentKey),
          ContentLanguage.es,
        ),
      );
    } catch (_) {
      // Keep defaults if storage fails.
    }
  }

  Future<void> setApp(AppLanguage value) async {
    _changedByUser = true;
    state = state.copyWith(app: value);
    try {
      final prefs = await SharedPreferences.getInstance();
      await prefs.setString(_appKey, value.name);
    } catch (_) {}
  }

  Future<void> setContent(ContentLanguage value) async {
    if (!value.available) return; // disabled options cannot be selected
    _changedByUser = true;
    state = state.copyWith(content: value);
    try {
      final prefs = await SharedPreferences.getInstance();
      await prefs.setString(_contentKey, value.name);
    } catch (_) {}
  }
}

import 'package:flutter/foundation.dart';

import '../errors/failure.dart';

@immutable
sealed class Result<S, F extends Failure> {
  const Result();

  bool get isSuccess => this is Ok<S, F>;

  bool get isFailure => this is Err<S, F>;

  R fold<R>({
    required R Function(S value) onSuccess,
    required R Function(F failure) onFailure,
  }) {
    return switch (this) {
      Ok<S, F>(:final value) => onSuccess(value),
      Err<S, F>(:final failure) => onFailure(failure),
    };
  }
}

final class Ok<S, F extends Failure> extends Result<S, F> {
  const Ok(this.value);

  final S value;
}

final class Err<S, F extends Failure> extends Result<S, F> {
  const Err(this.failure);

  final F failure;
}

import 'dart:math' as math;

// LEARN (parsing messy data): the API sends ki in many shapes:
//   "60.000.000"   dots are THOUSANDS separators (Spanish style)
//   "19.84 Septillion"   dot is a DECIMAL because a scale word follows
//   "90 Septillion" / "40 septillion"   different casing
//   "42.000 "      trailing space
//   "0"
// One pure function turns all of them into a double. Pure functions (no
// widgets, no state) are the easiest code to unit test: see
// test/dragonball_logic_test.dart.

const Map<String, int> _scaleWords = {
  'thousand': 3,
  'million': 6,
  'billion': 9,
  'trillion': 12,
  'quadrillion': 15,
  'quintillion': 18,
  'sextillion': 21,
  'septillion': 24,
  'octillion': 27,
  'nonillion': 30,
  'decillion': 33,
};

/// Bars are drawn on a log scale from 10^0 up to 10^30.
const double kMaxKiExponent = 30;

/// Returns null when the text is not a ki value we understand.
double? parseKi(String raw) {
  final text = raw.trim().toLowerCase();
  if (text.isEmpty) return null;

  // number part (digits, dots, commas) + optional scale word
  final match = RegExp(r'^([\d.,]+)\s*([a-z]+)?$').firstMatch(text);
  if (match == null) return null;

  final numberPart = match.group(1)!;
  final word = match.group(2);

  if (word != null) {
    final exponent = _scaleWords[word];
    if (exponent == null) return null;
    final base = double.tryParse(numberPart.replaceAll(',', ''));
    if (base == null) return null;
    return base * math.pow(10, exponent).toDouble();
  }

  // No scale word -> dots/commas are thousands separators.
  return double.tryParse(numberPart.replaceAll(RegExp(r'[.,]'), ''));
}

/// The bigger of current ki and max ki, used for sorting and tiers.
double powerOf(String ki, String maxKi) =>
    math.max(parseKi(ki) ?? 0, parseKi(maxKi) ?? 0);

/// Maps a ki value to 0..1 on a LOGARITHMIC scale.
/// Why: 60 million vs 90 septillion differ by 10^17. On a linear bar the
/// small one would be invisible. On a log bar both are readable.
double kiScale(double? value) {
  if (value == null || value <= 1) return 0;
  final exponent = math.log(value) / math.ln10;
  return (exponent / kMaxKiExponent).clamp(0.0, 1.0).toDouble();
}

const List<String> _suffixes = [
  'K', 'M', 'B', 'T', 'Qa', 'Qi', 'Sx', 'Sp', 'Oc', 'No',
];

String _trimZeros(String s) =>
    s.contains('.') ? s.replaceFirst(RegExp(r'\.?0+$'), '') : s;

/// 60000000 -> "60M", 9e25 -> "90Sp". Compact text for small UI spaces.
String formatKi(double? value) {
  if (value == null) return '?';
  if (value < 1000) return value.toStringAsFixed(0);

  // The tiny epsilon protects against log10(1e24) = 23.999999999.
  final exponent = (math.log(value) / math.ln10 + 1e-9).floor();
  final group = math.min(math.max(exponent ~/ 3, 1), _suffixes.length);
  final scaled = value / math.pow(10, group * 3);

  final digits = scaled >= 100
      ? scaled.toStringAsFixed(0)
      : scaled >= 10
          ? scaled.toStringAsFixed(1)
          : scaled.toStringAsFixed(2);

  return '${_trimZeros(digits)}${_suffixes[group - 1]}';
}

/// Power tiers. Names are translated in AppStrings.tierName.
enum KiTier { civilian, warrior, elite, legendary, mythic, divine }

KiTier tierFor(double power) {
  if (power < 1e3) return KiTier.civilian;
  if (power < 1e6) return KiTier.warrior;
  if (power < 1e9) return KiTier.elite;
  if (power < 1e15) return KiTier.legendary;
  if (power < 1e21) return KiTier.mythic;
  return KiTier.divine;
}

const Map<String, String> _knownForms = {
  'ssj': 'Super Saiyan',
  'ssj2': 'Super Saiyan 2',
  'ssj3': 'Super Saiyan 3',
  'ssj4': 'Super Saiyan 4',
  'ssjb': 'Super Saiyan Blue',
  'ssjg': 'Super Saiyan God',
  'ultrainstinc': 'Ultra Instinct', // typo exists in the API itself
  'ultrainstinct': 'Ultra Instinct',

/// "Goku SSJB" + "Goku" -> "Super Saiyan Blue".
/// Steps: drop the repeated character name, then map abbreviations.
String prettyTransformationName(String raw, String characterName) {
  var name = raw.trim();
  final prefix = characterName.trim();

  if (prefix.isNotEmpty && name.toLowerCase().startsWith(prefix.toLowerCase())) {
    name = name.substring(prefix.length).trim();
  }
  if (name.isEmpty) return raw.trim();

  final key = name.toLowerCase().replaceAll(' ', '');
  return _knownForms[key] ?? name;
}

  Future<Page<T>?> _readCachedFlexibleList<T>({
    required Uri uri,
    required int currentPage,
    required T Function(Map<String, dynamic>) parser,
  }) async {
    try {
      final cached = await _cache.read(uri.toString());

      if (cached == null) return null;

      final decoded = jsonDecode(cached.body);

      if (decoded is List) {
        final items = decoded
            .map((entry) => parser(_toMap(entry)))
            .toList();

        return Page<T>(
          items: List.unmodifiable(items),
          currentPage: 1,
          totalPages: 1,
          hasNext: false,
        );
      }

      if (decoded is Map) {
        final body = _toMap(decoded);

        final rawItems = body['items'];
        final meta = body['meta'] is Map
            ? _toMap(body['meta'])
            : <String, dynamic>{};
        final links = body['links'] is Map
            ? _toMap(body['links'])
            : <String, dynamic>{};

        final items = rawItems is List
            ? rawItems.map((entry) => parser(_toMap(entry))).toList()
            : <T>[];

        final current = _toInt(meta['currentPage'], currentPage);
        final total = _toInt(meta['totalPages'], 1);
        final nextLink = links['next']?.toString() ?? '';

        return Page<T>(
          items: List.unmodifiable(items),
          currentPage: current,
          totalPages: total,
          hasNext: nextLink.isNotEmpty || current < total,
        );
      }

      return null;
    } catch (_) {
      // Missing, corrupted, or incompatible cache is not fatal.
      return null;
    }
  }

      if (decoded is! Map) return null;

    } catch (_) {

      return null;
    }
  }



  /// Fetches a response from the API.
  ///
  /// Cache writes happen only after the response passes the expected
  /// JSON/model parsing in _fetchFlexibleList or _fetchSingle.
  ///
  /// Network failures are returned to the caller instead of being
  /// hidden by a cached response. The notifier can preserve existing
  /// cached UI data while also reporting the failure.
  Future<String> _requestBody(Uri uri) async {
    try {
      final response = await _client
          .get(uri)
          .timeout(const Duration(seconds: 20));

      if (response.statusCode < 200 || response.statusCode >= 300) {
        final status = response.statusCode;

        final message = switch (status) {
          400 => 'The request was invalid. Please check your filters.',
          401 => 'Authentication is required to access this resource.',
          403 => 'You do not have permission to access this resource.',
          404 => 'The requested resource was not found.',
          408 => 'The server timed out while processing the request.',
          429 => 'Too many requests. Please wait and try again.',
          >= 500 => 'The server encountered an error. Please try again.',
          _ => 'The server returned HTTP $status.',
        };

        throw _RepositoryRequestException(
          ServerFailure(message, status),
        );
      }

      return response.body;
    } on _RepositoryRequestException {
      rethrow;
    } on SocketException {
      throw const _RepositoryRequestException(
        NetworkFailure(
          'No internet connection. Check your connection and try again.',
        ),
      );
    } on TimeoutException {
      throw const _RepositoryRequestException(
        NetworkFailure(
          'The request timed out. Check your connection and try again.',
        ),
      );
    } on http.ClientException {
      throw const _RepositoryRequestException(
        NetworkFailure(
          'Could not connect to the server. Check your internet connection.',
        ),
      );
    } catch (error) {
      throw _RepositoryRequestException(
        UnknownFailure(
          'An unexpected network error occurred: $error',
        ),
      );
    }
  }


  Future<Result<Page<T>, Failure>> _fetchFlexibleList<T>({
    required Uri uri,
    required int currentPage,
    required T Function(Map<String, dynamic>) parser,
  }) async {
    try {
      final responseBody = await _requestBody(uri);
      final decoded = jsonDecode(responseBody);

      
      if (decoded is List) {
        final items = decoded
            .map((entry) => parser(_toMap(entry)))
            .toList();

        final page = Page<T>(
          items: List.unmodifiable(items),
          currentPage: 1,
          totalPages: 1,
          hasNext: false,
        );

        await _cache.write(uri.toString(), responseBody);

        return Ok(page);
      }


      
      if (decoded is Map) {
        final body = _toMap(decoded);

        final rawItems = body['items'];
        final meta = body['meta'] is Map
            ? _toMap(body['meta'])
            : <String, dynamic>{};
        final links = body['links'] is Map
            ? _toMap(body['links'])
            : <String, dynamic>{};

        if (rawItems is! List) {
          return const Err(
            ParseFailure(
              'The API response does not contain a valid items list.',
            ),
          );
        }

        final items = rawItems
            .map((entry) => parser(_toMap(entry)))
            .toList();

        final current = _toInt(meta['currentPage'], currentPage);
        final total = _toInt(meta['totalPages'], 1);
        final nextLink = links['next']?.toString() ?? '';

        final page = Page<T>(
          items: List.unmodifiable(items),
          currentPage: current,
          totalPages: total,
          hasNext: nextLink.isNotEmpty || current < total,
        );

        await _cache.write(uri.toString(), responseBody);

        return Ok(page);
      }


      return const Err(
        ParseFailure('The API returned an unexpected JSON shape.'),
      );
    } on _RepositoryRequestException catch (error) {
      return Err(error.failure);
    } on FormatException {
      return const Err(
        ParseFailure('The API response was not valid JSON.'),
      );
    } catch (error) {
      return Err(
        UnknownFailure('Could not process the response: $error'),
      );
    }
  }


  Future<Result<T, Failure>> _fetchSingle<T>(
    Uri uri,
    T Function(Map<String, dynamic>) parser,
  ) async {
    try {
      final responseBody = await _requestBody(uri);
      final decoded = jsonDecode(responseBody);

      if (decoded is! Map) {
        return const Err(
          ParseFailure('The API returned an unexpected JSON shape.'),
        );
      }

      // 1. Parse and validate the response first.
      final value = parser(_toMap(decoded));

      // 2. Cache only after parsing succeeds.
      await _cache.write(uri.toString(), responseBody);

      // 3. Return the validated model.
      return Ok(value);
    } on _RepositoryRequestException catch (error) {
      return Err(error.failure);
    } on FormatException {
      return const Err(
        ParseFailure('The API response was not valid JSON.'),
      );
    } catch (error) {
      return Err(
        UnknownFailure('Could not process the response: $error'),
      );
    }
  }


  Map<String, dynamic> _toMap(dynamic value) {
    if (value is Map<String, dynamic>) return value;

    if (value is Map) {
      return value.map(
        (key, value) => MapEntry(key.toString(), value),
      );
    }

    throw const FormatException('Expected a JSON object.');
  }

  int _toInt(dynamic value, int fallback) {
    if (value is int) return value;

    return int.tryParse(value?.toString() ?? '') ?? fallback;
  }
}

import 'dart:convert';

class CachedJsonResponse {
  const CachedJsonResponse({
    required this.body,
    required this.cachedAt,
  });

  final String body;
  final DateTime cachedAt;

  bool isFresh(Duration maxAge) {
    final age = DateTime.now().difference(cachedAt);

    return !age.isNegative && age <= maxAge;
  }
}

class JsonResponseCacheStats {
  const JsonResponseCacheStats({
    required this.fileCount,
    required this.totalBytes,
  });

  final int fileCount;
  final int totalBytes;
}

class JsonResponseCache {
  JsonResponseCache._();

  static final JsonResponseCache instance = JsonResponseCache._();

  static const Duration defaultMaxAge = Duration(hours: 24);

  Directory? _cacheDirectory;

  Future<JsonResponseCacheStats> stats() async {
    try {
      final directory = await _getCacheDirectory();

      if (!await directory.exists()) {
        return const JsonResponseCacheStats(
          fileCount: 0,
          totalBytes: 0,
        );
      }

      var count = 0;
      var bytes = 0;

      await for (final entity in directory.list(followLinks: false)) {
        if (entity is! File || !entity.path.endsWith('.json')) {
          continue;
        }

        try {
          bytes += await entity.length();
          count++;
        } catch (_) {
          // Ignore a file that disappears during the scan.
        }
      }

      return JsonResponseCacheStats(
        fileCount: count,
        totalBytes: bytes,
      );
    } catch (_) {
      return const JsonResponseCacheStats(
        fileCount: 0,
        totalBytes: 0,
      );
    }
  }
  
  Future<Directory> _getCacheDirectory() async {
    final existing = _cacheDirectory;

    if (existing != null) {
      return existing;
    }

    final appDirectory = await getApplicationSupportDirectory();

    final directory = Directory(
      '${appDirectory.path}${Platform.pathSeparator}'
      'dragonball_api_cache',
    );

    await directory.create(recursive: true);

    _cacheDirectory = directory;
    return directory;
  }

  String _fileName(String key) {
    final hash = sha256.convert(utf8.encode(key)).toString();
    return '$hash.json';
  }

  Future<File> _getFile(String key) async {
    final directory = await _getCacheDirectory();

    return File(
      '${directory.path}${Platform.pathSeparator}'
      '${_fileName(key)}',
    );
  }

  /// Reads a cached response, including stale responses.
  ///
  /// Check [CachedJsonResponse.isFresh] to determine whether
  /// the response is within its freshness period.
  Future<CachedJsonResponse?> read(String key) async {
    try {
      final file = await _getFile(key);

      if (!await file.exists()) {
        return null;
      }

      final contents = await file.readAsString();
      final envelope = jsonDecode(contents);

      if (envelope is! Map<String, dynamic>) {
        return null;
      }

      final body = envelope['body'];
      final cachedAtMillis = envelope['cachedAt'];

      if (body is! String || cachedAtMillis is! int) {
        return null;
      }

      // Reject corrupted cached JSON.
      jsonDecode(body);

      return CachedJsonResponse(
        body: body,
        cachedAt: DateTime.fromMillisecondsSinceEpoch(
          cachedAtMillis,
        ),
      );
    } on FileSystemException {
      return null;
    } on FormatException {
      return null;
    } catch (_) {
      // A cache failure must not crash the application.
      return null;
    }
  }

  /// Writes a successful API response to disk.
  Future<void> write(String key, String body) async {
    try {
      // Only persist valid JSON.
      jsonDecode(body);

      final file = await _getFile(key);

      final envelope = jsonEncode({
        'cachedAt': DateTime.now().millisecondsSinceEpoch,
        'body': body,
      });

      await file.writeAsString(
        envelope,
        flush: true,
      );
    } catch (_) {
      // Cache failures must not turn a successful API response
      // into a failed request.
    }
  }

  Future<void> delete(String key) async {
    try {
      final file = await _getFile(key);

      if (await file.exists()) {
        await file.delete();
      }
    } catch (_) {
      // Deleting cache is best-effort.
    }
  }

  Future<void> clear() async {
    final directory = _cacheDirectory;

    try {
      final target = directory ?? await _getCacheDirectory();

      if (await target.exists()) {
        await target.delete(recursive: true);
      }
    } catch (_) {
      // Cache cleanup is best-effort.
    } finally {
      _cacheDirectory = null;
    }
  }
}

    final rawPlanet = json['originPlanet'];

    return CharacterProfile(
      id: _readInt(json['id']),
      name: _readString(json['name'], fallback: 'Unknown'),
      ki: _readString(json['ki'], fallback: '0'),
      maxKi: _readString(json['maxKi'], fallback: '0'),
      race: _readString(json['race'], fallback: 'Unknown Race'),
      gender: _readString(json['gender'], fallback: 'Unknown'),
      description: _readString(json['description']),
      imageUrl: _readString(json['image']),
      affiliation: _readString(json['affiliation'], fallback: 'None'),
      originPlanet: rawPlanet is Map
          ? Planet.fromJson(_asMap(rawPlanet))
          : null,
      transformations: List.unmodifiable(transformations),
    );
  }

class AppSettingsController extends Notifier<AppSettings> {
  static const _themeKey = 'app_theme_mode';
  static const _accentKey = 'app_accent_color';
  static const _animationsKey = 'app_animations_enabled';

  static const _charactersGridKey = 'characters_grid_view';
  static const _planetsGridKey = 'planets_grid_view';
  static const _favoritesGridKey = 'favorites_grid_view';

  static const _compactCardsKey = 'app_compact_cards';
  static const _catalogSortKey = 'app_catalog_sort';
  static const _favoritesSortKey = 'app_favorites_sort';
  static const _defaultTabKey = 'app_default_tab';
  static const _textScaleKey = 'app_text_scale';
  static const _reducedMotionKey = 'app_reduced_motion';
  static const _highContrastKey = 'app_high_contrast';

  bool _restoreStarted = false;
  bool _userChangedSettings = false;

  @override
  AppSettings build() {
    if (!_restoreStarted) {
      _restoreStarted = true;
      _restore();
    }

    return const AppSettings();
  }

  Future<void> _restore() async {
    try {
      final prefs = await SharedPreferences.getInstance();

      final restored = AppSettings(
        isLoaded: true,
        themeMode: switch (prefs.getString(_themeKey)) {
          'light' => ThemeMode.light,
          'dark' => ThemeMode.dark,
          _ => ThemeMode.system,
        },
        accentColor: _validAccent(prefs.getString(_accentKey)),
        animationsEnabled:
            prefs.getBool(_animationsKey) ?? true,
        charactersGridView:
            prefs.getBool(_charactersGridKey) ?? true,
        planetsGridView:
            prefs.getBool(_planetsGridKey) ?? true,
        favoritesGridView:
            prefs.getBool(_favoritesGridKey) ?? true,
        compactCards: prefs.getBool(_compactCardsKey) ?? false,
        catalogSortOrder: _validCatalogSort(
          prefs.getString(_catalogSortKey),
        ),
        favoritesSortOrder: _validFavoritesSort(
          prefs.getString(_favoritesSortKey),
        ),
        defaultTab: _validTab(prefs.getInt(_defaultTabKey)),
        textScale: _validTextScale(prefs.getDouble(_textScaleKey)),
        reducedMotion:
            prefs.getBool(_reducedMotionKey) ?? false,
        highContrast: prefs.getBool(_highContrastKey) ?? false,
      );

      // Don't overwrite choices made while storage was loading.
      if (!_userChangedSettings) {
        state = restored;
      } else {
        state = state.copyWith(isLoaded: true);
      }
    } catch (_) {
      state = state.copyWith(isLoaded: true);
    }
  }

  String _validAccent(String? value) =>
      const {'orange', 'blue', 'green', 'purple'}.contains(value)
          ? value!
          : 'orange';

  String _validCatalogSort(String? value) =>
      const {'default', 'name_asc', 'name_desc'}.contains(value)
          ? value!
          : 'default';

  String _validFavoritesSort(String? value) =>
      const {'recent', 'id_asc', 'id_desc'}.contains(value)
          ? value!
          : 'recent';

  int _validTab(int? value) =>
      value != null && value >= 0 && value <= 3 ? value : 0;

  double _validTextScale(double? value) =>
      value != null &&
              const [0.9, 1.0, 1.15, 1.3].contains(value)
          ? value
          : 1.0;

  Future<void> _save(String key, Object value) async {
    try {
      final prefs = await SharedPreferences.getInstance();

      if (value is String) {
        await prefs.setString(key, value);
      } else if (value is bool) {
        await prefs.setBool(key, value);
      } else if (value is int) {
        await prefs.setInt(key, value);
      } else if (value is double) {
        await prefs.setDouble(key, value);
      }
    } catch (_) {
      // Keep the in-memory setting even if persistence fails.
    }
  }

  void setThemeMode(ThemeMode value) {
    _change(state.copyWith(themeMode: value));
    _save(
      _themeKey,
      switch (value) {
        ThemeMode.light => 'light',
        ThemeMode.dark => 'dark',
        ThemeMode.system => 'system',
      },
    );
  }

  void setAccentColor(String value) {
    if (!const {'orange', 'blue', 'green', 'purple'}.contains(value)) {
      return;
    }
    _change(state.copyWith(accentColor: value));
    _save(_accentKey, value);
  }

  void setAnimationsEnabled(bool value) {
    _change(state.copyWith(animationsEnabled: value));
    _save(_animationsKey, value);
  }

  void setCharactersGridView(bool value) {
    _change(state.copyWith(charactersGridView: value));
    _save(_charactersGridKey, value);
  }

  void setPlanetsGridView(bool value) {
    _change(state.copyWith(planetsGridView: value));
    _save(_planetsGridKey, value);
  }

  void setFavoritesGridView(bool value) {
    _change(state.copyWith(favoritesGridView: value));
    _save(_favoritesGridKey, value);
  }

  void setCompactCards(bool value) {
    _change(state.copyWith(compactCards: value));
    _save(_compactCardsKey, value);
  }

  void setCatalogSortOrder(String value) {
    if (_validCatalogSort(value) != value) return;
    _change(state.copyWith(catalogSortOrder: value));
    _save(_catalogSortKey, value);
  }

  void setFavoritesSortOrder(String value) {
    if (_validFavoritesSort(value) != value) return;
    _change(state.copyWith(favoritesSortOrder: value));
    _save(_favoritesSortKey, value);
  }

  void setDefaultTab(int value) {
    if (value < 0 || value > 3) return;
    _change(state.copyWith(defaultTab: value));
    _save(_defaultTabKey, value);
  }

  void setTextScale(double value) {
    if (!_validTextScale(value).isFinite ||
        _validTextScale(value) != value) {
      return;
    }
    _change(state.copyWith(textScale: value));
    _save(_textScaleKey, value);
  }

  void setReducedMotion(bool value) {
    _change(state.copyWith(reducedMotion: value));
    _save(_reducedMotionKey, value);
  }

  void setHighContrast(bool value) {
    _change(state.copyWith(highContrast: value));
    _save(_highContrastKey, value);
  }

  void _change(AppSettings value) {
    _userChangedSettings = true;
    state = value;
  }

  Future<void> resetPreferences() async {
    _userChangedSettings = true;
    state = const AppSettings(isLoaded: true);

    try {
      final prefs = await SharedPreferences.getInstance();
      for (final key in [
        _themeKey,
        _accentKey,
        _animationsKey,
        _charactersGridKey,
        _planetsGridKey,
        _favoritesGridKey,
        _compactCardsKey,
        _catalogSortKey,
        _favoritesSortKey,
        _defaultTabKey,
        _textScaleKey,
        _reducedMotionKey,
        _highContrastKey,
      ]) {
        await prefs.remove(key);
      }
    } catch (_) {}
  }
}

class CharacterProfileView extends ConsumerStatefulWidget {
  const CharacterProfileView({
    super.key,
    required this.characterId,
    this.fromFavorites = false, // kept so existing favorites code still works
    this.nav,
  });

  final int characterId;
  final bool fromFavorites;

  ConsumerState<CharacterProfileView> createState() =>

class _CharacterProfileViewState extends ConsumerState<CharacterProfileView> {
  // `late final` + initializer runs on first use, when `widget` is available.
  late final CharacterNavContext _nav = widget.nav ?? _fallbackNav();
  late final PageController _pages = PageController(initialPage: _nav.index);
  late int _index = _nav.index;

  CharacterNavContext _fallbackNav() => CharacterNavContext(
        source: widget.fromFavorites ? const FromFavorites() : const FromAll(),
        siblingIds: [widget.characterId],
        index: 0,
        heroTagFor:
            widget.fromFavorites ? favoriteCharacterHeroTag : characterHeroTag,
      );

  @override
  void dispose() {
    _pages.dispose();
    super.dispose();
  }

  @override
  Widget build(BuildContext context) {
    final s = ref.watch(stringsProvider);
    final ids = _nav.siblingIds;
    final currentId = ids[_index];

    // Watch only the NAME of the current character, so the breadcrumb updates
    // when the profile loads or when the user swipes.
    final currentName = ref.watch(
      characterProfileProvider(currentId).select((a) => a.asData?.value.name),
    );

    final labels = _nav.source.labels(s);
    final crumbs = <Crumb>[
      for (var i = 0; i < labels.length; i++)
        Crumb(
          labels[i],
          onTap: () => _nav.source.onCrumbTap(i, ref, context),
        ),
      Crumb(currentName ?? '…'),
    ];

    return Scaffold(
      appBar: AppBar(
        title: Text(s.characterDossier),
        actions: [
          FavoriteHeart(characterId: currentId, overlay: false),
          const SizedBox(width: AppSpace.x1),
        ],
      ),
      body: Column(
        children: [
          BreadcrumbBar(
            crumbs: crumbs,
            trailing: ids.length > 1

                characterId: ids[i],
                heroTag: _nav.heroTagFor(ids[i]),
              ),
            ),
          ),
        ],
      ),
    );
  }
}

          key: ValueKey<int>(character.id),
          character: character,
          heroTag: heroTag,

// LEARN (keeping TWO screens in sync): the full-screen viewer and this strip
// share ONE ValueNotifier<int> (the viewer's page index).
//   * Swiping in the viewer writes to the notifier.
//   * This strip LISTENS and scrolls itself to the same form, while it is
//     hidden behind the viewer. When the viewer closes (X button, system back
//     or back swipe) the Hero finds the right card already on screen and the
//     image flies back to exactly the form you were looking at.
//   * The viewer's onBeforePop is NOT called on system back, so listening to
//     the notifier is more reliable than a "before pop" callback.
// The selected card also gets a colored border and a slight scale, so you can
// see which form is active.
class _TransformationStrip extends StatefulWidget {
  const _TransformationStrip({
    required this.characterId,
    required this.characterName,
    required this.transformations,
    required this.fallbackColor,
  });

  final int characterId;
  final String characterName;
  final List<Transformation> transformations;
  final Color fallbackColor;

  @override
  State<_TransformationStrip> createState() => _TransformationStripState();
}

class _TransformationStripState extends State<_TransformationStrip> {
  static const double _itemWidth = 150;
  static const double _gap = AppSpace.x3;

  final ScrollController _scroll = ScrollController();

  // The viewer's PageView is endless (page = count * 1000 + index), so this
  // holds a PAGE number; the form index is always page % count.
  late final ValueNotifier<int> _page;

  bool _viewerOpen = false;

  int get _count => widget.transformations.length;

  @override
  void initState() {
    super.initState();
    _page = ValueNotifier<int>(_count * 1000);
    _page.addListener(_followViewer);
  }

  @override
  void dispose() {
    _page.removeListener(_followViewer);
    _page.dispose();
    _scroll.dispose();
    super.dispose();
  }

  String _heroTag(int index) => 'form-${widget.characterId}-$index';

  // Runs every time the viewer changes page.
  void _followViewer() {
    if (!_viewerOpen || _count == 0) return;
    _centerOn(_page.value % _count);
  }

  // jumpTo (not animateTo): the strip is covered by the viewer, so animating
  // would only risk a half-finished scroll when the Hero flight starts.
  void _centerOn(int index) {
    if (!_scroll.hasClients) return;

    final position = _scroll.position;
    final target =
        index * (_itemWidth + _gap) - (position.viewportDimension - _itemWidth) / 2;

    _scroll.jumpTo(
      target.clamp(position.minScrollExtent, position.maxScrollExtent).toDouble(),
    );
  }

  Future<void> _open(int index, List<String> names) async {
    // Start from the tapped form, even if an earlier session swiped far away.
    _page.value = _count * 1000 + index;
    _viewerOpen = true;

    await Navigator.of(context).push(
      MaterialPageRoute<void>(
        builder: (_) => FullScreenImageViewer(
          images: [
            for (var j = 0; j < _count; j++)
              FullScreenImageItem(
                imageUrl: widget.transformations[j].imageUrl,
                heroTag: _heroTag(j),
                title: names[j],
              ),
          ],
          initialIndex: index,
          // With ONE image the viewer has no PageView; passing the notifier
          // would make it look for page 1000 and drop the Hero.
          pageNotifier: _count > 1 ? _page : null,
        ),
      ),
    );

    _viewerOpen = false;
  }

  @override
  Widget build(BuildContext context) {
    final theme = Theme.of(context);
    final transformations = widget.transformations;

    // Clean every name (e.g. "Goku SSJB" -> "Super Saiyan Blue").
    final names = [
      for (final t in transformations)
        prettyTransformationName(t.name, widget.characterName),
    ];

    return SizedBox(
      height: 240,
      // Rebuilds only the strip when the selected form changes.
      child: ValueListenableBuilder<int>(
        valueListenable: _page,
        builder: (context, page, _) {
          final selected = page % _count;

          return ListView.separated(
            controller: _scroll,
            scrollDirection: Axis.horizontal,
            clipBehavior: Clip.none,
            padding: const EdgeInsets.symmetric(vertical: AppSpace.x2),
            itemCount: _count,
            separatorBuilder: (_, __) => const SizedBox(width: _gap),
            itemBuilder: (context, i) {
              final t = transformations[i];
              final accent = formAccentColor(names[i], widget.fallbackColor);
              final isSelected = i == selected;

              return AnimatedScale(
                scale: isSelected ? 1.0 : 0.96,
                duration: AppMotion.base,
                curve: AppMotion.curve,
                child: SizedBox(
                  width: _itemWidth,
                  child: Card(
                    shape: RoundedRectangleBorder(
                      borderRadius: AppRadius.lgAll,
                      side: BorderSide(
                        color: isSelected
                            ? accent
                            : theme.colorScheme.outlineVariant,
                        width: isSelected ? 2 : 1,
                      ),
                    ),
                    child: InkWell(
                      onTap: () => _open(i, names),
                      child: Column(
                        crossAxisAlignment: CrossAxisAlignment.stretch,
                        children: [
                          Container(height: 4, color: accent),
                          Expanded(
                            child: Padding(
                              padding: const EdgeInsets.all(AppSpace.x2),
                              child: Hero(
                                tag: _heroTag(i),
                                child: buildCharacterImage(
                                  t.imageUrl,
                                  fit: BoxFit.contain,
                                  showLoadingIndicator: true,
                                  errorWidget: const Icon(AppIcons.imageBroken),
                                ),
                              ),
                            ),
                          ),
                          Padding(
                            padding: const EdgeInsets.fromLTRB(
                              AppSpace.x2,
                              0,
                              AppSpace.x2,
                              AppSpace.x3,
                            ),
                            child: Column(
                              children: [
                                Text(
                                  names[i],
                                  maxLines: 2,
                                  textAlign: TextAlign.center,
                                  overflow: TextOverflow.ellipsis,
                                  style: theme.textTheme.titleSmall,
                                ),
                                const SizedBox(height: AppSpace.x1),
                                Row(
                                  mainAxisAlignment: MainAxisAlignment.center,
                                  children: [
                                    Icon(AppIcons.ki, size: 14, color: accent),
                                    const SizedBox(width: 4),
                                    Text(
                                      formatKi(parseKi(t.ki)),
                                      style: theme.textTheme.labelMedium,
                                    ),
                                  ],
                                ),
                              ],
                            ),
                          ),
                        ],
                      ),
                    ),
                  ),
                ),
              );
            },
          );
        },
      ),
    );
  }
}

  @override
  Widget build(BuildContext context, WidgetRef ref) {
    // Local UI state: it does not need to be shared across the application.
    final characterQuery = useState('');
    final planetQuery = useState('');

    final characterSearchFocusNode = useFocusNode();
    final planetSearchFocusNode = useFocusNode();

    final characterSearchController = useTextEditingController();
    final planetSearchController = useTextEditingController();

    final planetDebouncer = useMemoized(SearchDebouncer.new);

    // Register and clean up scroll listeners with the widget lifecycle.
    useEffect(() {
      void onCharacterScroll() {
        if (characterSearchFocusNode.hasFocus) {
          characterSearchFocusNode.unfocus();
        }

        final characterError = newState.characterError;
        final characterErrorChanged =

        }

        final planetError = newState.planetError;
        final planetErrorChanged =
            planetError != null &&
            planetError != oldState?.planetError;

        if (planetErrorChanged && newState.planets.isNotEmpty) {
          messages.add('Planets: $planetError');

        final messenger = ScaffoldMessenger.of(context);

        messenger
          ..hideCurrentSnackBar()
          ..showSnackBar(
            SnackBar(
              content: Text(messages.join('\n')),
            ),
          );
      },
    );


    final catalogAsync = ref.watch(dragonBallControllerProvider);

    // Only the count is selected from favorites for the app bar.
    final favoriteCount = ref.watch(
      dragonBallFavoritesProvider.select((ids) => ids.length),
    );

    final s = ref.watch(stringsProvider);

    final titles = <(String, String)>[
      ('Warrior database', 'Explore fighters and villains'),
      (s.exploreTitle, s.exploreSubtitle),
      ('Cosmic atlas', 'Explore worlds across the universe'),
      ('Favorite warriors', 'Your saved characters'),
    ];

    final (title, subtitle) = titles[selectedTab.value];

    return Scaffold(
      appBar: AppBar(
        title: Column(
          crossAxisAlignment: CrossAxisAlignment.start,
          children: [
            Text(
              title,
              style: Theme.of(context).textTheme.titleLarge,
            ),
            Text(
              subtitle,
              style: Theme.of(context).textTheme.bodySmall,
            ),
          ],
        ),
        actions: [
          Tooltip(
            message: 'Favorite characters in the current catalogue',
            child: Padding(
              padding: const EdgeInsets.only(right: 4),
              child: Center(
                child: Chip(
                  avatar: const Icon(
                    Icons.favorite,
                    color: Colors.red,
                    size: 16,
                  ),
                  label: Text('$favoriteCount'),
                  visualDensity: VisualDensity.compact,
                ),
              ),
            ),
          ),
          IconButton(
            tooltip: 'Settings',
            onPressed: () {
              Navigator.of(context).push(
                MaterialPageRoute<void>(
                  builder: (_) => const SettingsScreen(),
                ),
              );
            },
            icon: const Icon(Icons.settings_outlined),
          ),
          const SizedBox(width: 4),
        ],
      ),
      bottomNavigationBar: NavigationBar(
        selectedIndex: selectedTab.value,
        onDestinationSelected: (index) {
          selectedTab.value = index;
        },
        destinations: [
          const NavigationDestination(
            icon: Icon(Icons.people_outline),
            selectedIcon: Icon(Icons.people),
            label: 'Characters',
          ),
          NavigationDestination(
            icon: const Icon(AppIcons.explore),
            selectedIcon: const Icon(AppIcons.exploreFill),
            label: s.tabExplore,
          ),
          const NavigationDestination(
            icon: Icon(Icons.public_outlined),
            selectedIcon: Icon(Icons.public),
            label: 'Planets',
          ),
          const NavigationDestination(
            icon: Icon(Icons.favorite_border),
            selectedIcon: Icon(Icons.favorite),
            label: 'Favorites',
          ),
        ],
      ),
      body: catalogAsync.when(
        loading: () => const Center(
          child: CircularProgressIndicator.adaptive(),
        ),
        error: (error, _) => _FullErrorView(
          message: error.toString(),
          onRetry: () {
            ref.invalidate(dragonBallControllerProvider);
          },
        ),
        data: (state) => IndexedStack(
          index: selectedTab.value,
          children: [
            _CharacterTab(
              state: state,
              isGridView: characterGridView,
              sortOrder: settings.catalogSortOrder,
              searchFocusNode: characterSearchFocusNode,
              searchController: characterSearchController,
              searchQuery: characterQuery.value,
              scrollController: characterScrollController,
              onSearchChanged: (query) {
                characterQuery.value = query;
                characterDebouncer.run(
                  const Duration(milliseconds: 350),
                  () {
                    ref
                        .read(dragonBallControllerProvider.notifier)
                        .applyCharacterFilters(name: query);
                  },
                );
              },
              onOpenFilters: () {
                _showCharacterFilterSheet(
                  context,
                  ref,
                  onResetSearch: () {
                    characterDebouncer.cancel();
                    characterSearchController.clear();
                    characterQuery.value = '';
                  },
                );
              },
              onRetry: () => ref
                  .read(dragonBallControllerProvider.notifier)
                  .applyCharacterFilters(),
              
              onReset: () {
                characterDebouncer.cancel();
                characterSearchController.clear();
                characterQuery.value = '';

                ref
                    .read(dragonBallControllerProvider.notifier)
                    .resetCharacterFilters();
              },
            ),
            const ExploreScreen(),
            _PlanetTab(
              state: state,
              isGridView: planetGridView,
              sortOrder: settings.catalogSortOrder,
              searchFocusNode: planetSearchFocusNode,
              searchController: planetSearchController,
              searchQuery: planetQuery.value,
              scrollController: planetScrollController,
              onSearchChanged: (query) {
                planetQuery.value = query;
                planetDebouncer.run(
                  const Duration(milliseconds: 350),
                  () {
                    ref
                        .read(dragonBallControllerProvider.notifier)
                        .applyPlanetFilters(name: query);
                  },
                );
              },
              onDestroyedChanged: (bool? destroyed) {
                final controller =
                    ref.read(dragonBallControllerProvider.notifier);

                if (destroyed == null) {
                  controller.applyPlanetFilters(clearDestroyedFilter: true);
                } else {
                  controller.applyPlanetFilters(isDestroyedFilter: destroyed);
                }
              },
              onRetry: () => ref
                  .read(dragonBallControllerProvider.notifier)
                  .applyPlanetFilters(),
              onReset: () {
                planetDebouncer.cancel();
                planetSearchController.clear();
                planetQuery.value = '';

                ref
                    .read(dragonBallControllerProvider.notifier)
                    .resetPlanetFilters();
              },
            ),
            _FavoritesTab(
              isGridView: favoritesGridView,
              sortOrder: settings.favoritesSortOrder,
              onBrowseCharacters: () {
                selectedTab.value = 0;
              },
            ),
          ],
        ),
      ),
    );
  }

  
  
  void _showCharacterFilterSheet(
    BuildContext context,
    WidgetRef ref, {
    required VoidCallback onResetSearch,
  }) {
    final currentState = ref
        .read(dragonBallControllerProvider)
        .asData
        ?.value;

    if (currentState == null) return;

    final controller =
        ref.read(dragonBallControllerProvider.notifier);

    showCharacterFilterSheet(
      context: context,
      selectedRace: currentState.selectedRace,
      selectedGender: currentState.selectedGender,
      selectedAffiliation: currentState.selectedAffiliation,
      races: _races,
      genders: _genders,
      affiliations: _affiliations,
      onResetSearch: onResetSearch,
      onApply: ({
        required String race,
        required String gender,
        required String affiliation,
        required bool resetSearch,
      }) {
        controller.applyCharacterFilters(
          name: resetSearch ? '' : null,
          race: race,
          gender: gender,
          affiliation: affiliation,
        );
      },
    );
  }
}


class _CharacterTab extends StatelessWidget {
  const _CharacterTab({
    required this.state,
    required this.isGridView,
    required this.searchController,
    required this.searchQuery,
    required this.scrollController,
    required this.onSearchChanged,
    required this.onOpenFilters,
    required this.onRetry,
    required this.onReset,
    required this.searchFocusNode,
    required this.sortOrder,
  });

  final DragonBallState state;
  final bool isGridView;
  final TextEditingController searchController;
  final String searchQuery;
  final ScrollController scrollController;
  final ValueChanged<String> onSearchChanged;
  final VoidCallback onOpenFilters;
  final Future<void> Function() onRetry;
  final VoidCallback onReset;
  final FocusNode searchFocusNode;
  final String sortOrder;


  @override
  Widget build(BuildContext context) {
  final displayedCharacters = [...state.characters];

  switch (sortOrder) {
    case 'name_asc':
      displayedCharacters.sort(
        (a, b) => a.name.toLowerCase().compareTo(
              b.name.toLowerCase(),
            ),
      );
      break;
    case 'name_desc':
      displayedCharacters.sort(
        (a, b) => b.name.toLowerCase().compareTo(
              a.name.toLowerCase(),
            ),
      );
      break;
  }

    return Column(
    children: [
      if (state.isLoadingCharacters)
        const LinearProgressIndicator(),      
      Padding(
        padding: const EdgeInsets.fromLTRB(16, 12, 16, 8),
        child: Row(
          children: [
            Expanded(
              child: TextField(
                controller: searchController,
                focusNode: searchFocusNode,
                onChanged: onSearchChanged,
                decoration: InputDecoration(
                  hintText: 'Search warriors...',
                  prefixIcon: const Icon(Icons.search),
                  suffixIcon: searchQuery.isEmpty
                      ? null
                      : IconButton(
                          tooltip: 'Clear search',
                          onPressed: () {
                            searchController.clear();
                            onSearchChanged('');
                          },
                          icon: const Icon(Icons.close),
                        ),
                ),
              ),
            ),
            const SizedBox(width: 8),
            Badge(
              isLabelVisible: state.hasActiveCharacterFilters,
              child: IconButton.filledTonal(
                tooltip: 'Filter characters',
                onPressed: onOpenFilters,
                icon: const Icon(Icons.tune),
              ),
            ),
            if (searchQuery.isNotEmpty ||
                state.hasActiveCharacterFilters) ...[
              const SizedBox(width: 4),
              IconButton(
                tooltip: 'Clear filters',
                onPressed: onReset,
                icon: const Icon(Icons.filter_alt_off),
              ),
            ],
          ],
        ),
      ),
        Expanded(
          child: LayoutBuilder(
            builder: (context, constraints) {
              if (state.isLoadingCharacters && state.characters.isEmpty) {
                return CatalogSkeleton(
                  isGridView: isGridView,
                  columns: _gridColumns(constraints.maxWidth),
                  gridAspectRatio: 0.66,
                );
              }

              if (state.characterError != null &&
                  state.characters.isEmpty) {
                return _InlineError(
                  message: state.characterError!,
                  onRetry: onRetry,
                );
              }

              if (state.characters.isEmpty) {
                return _EmptyView(
                  icon: Icons.person_search,
                  title: 'No warriors found',
                  subtitle: 'Try another search or clear your filters.',
                  onAction: onReset,
                  actionLabel: 'Reset',
                );
              }

              final columns = _gridColumns(constraints.maxWidth);

              
            return CatalogResults<Character>(
                    items: displayedCharacters,
                    isGridView: isGridView,
                    columns: columns,
                    gridAspectRatio: 0.66,
                    controller: scrollController,
                    isLoadingMore: state.isLoadingMoreCharacters,
                    onRefresh: onRetry,
                    itemBuilder: (context, character) {
                      final key = ValueKey<int>(character.id);

                      return isGridView
                          ? _CharacterTile(
                              key: key,
                              character: character,
                            )
                          : _CharacterListTile(
                              key: key,
                              character: character,
                            );
                    },
            );

            },
          ),
        ),
      ],
    );
  }
}

class _FavoritesTab extends ConsumerWidget {
  const _FavoritesTab({
    required this.isGridView,
    required this.onBrowseCharacters,
    required this.sortOrder,
  });

  final bool isGridView;
  final VoidCallback onBrowseCharacters;
  final String sortOrder;

  @override
  Widget build(BuildContext context, WidgetRef ref) {
    final favoriteIds = ref.watch(dragonBallFavoritesProvider);
    final ids = favoriteIds.toList();

    switch (sortOrder) {
      case 'id_asc':
        ids.sort();
        break;
      case 'id_desc':
        ids.sort((a, b) => b.compareTo(a));
        break;
    }

    return Column(
      children: [
        Padding(
          padding: const EdgeInsets.fromLTRB(16, 12, 16, 8),
          child: Align(
            alignment: Alignment.centerLeft,
            child: Text(
              'Saved characters',
              style: Theme.of(context).textTheme.titleMedium?.copyWith(
                    fontWeight: FontWeight.w600,
                  ),
            ),
          ),
        ),
        Expanded(
          child: ids.isEmpty
            ? _EmptyView(
                icon: Icons.favorite_border,
                title: 'No favorites yet',
                subtitle:
                    'Tap the heart on a character card to save a warrior.',
                onAction: onBrowseCharacters,
                actionLabel: 'Browse characters',
                actionIcon: Icons.people_outline,
              )
            : isGridView
                ? LayoutBuilder(
                    builder: (context, constraints) {
                      final columns = _gridColumns(
                        constraints.maxWidth,
                      );

                      return GridView.builder(
                        padding: const EdgeInsets.all(12),
                        itemCount: ids.length,
                        gridDelegate:
                            SliverGridDelegateWithFixedCrossAxisCount(
                          crossAxisCount: columns,
                          crossAxisSpacing: 12,
                          mainAxisSpacing: 12,
                          childAspectRatio: 0.66,
                        ),
                        itemBuilder: (context, index) {
                          return _FavoriteCharacterTile(
                            characterId: ids[index],
                          );
                        },
                      );
                    },
                  )
                : ListView.builder(
                    padding: const EdgeInsets.all(12),
                    itemCount: ids.length,
                    itemBuilder: (context, index) {
                      return _FavoriteCharacterListTile(
                        characterId: ids[index],
                      );
                    },
                  ),
        ),
      ],
    );
  }

    final initialState = DragonBallState.initial();

    // Read both catalogues from disk before requesting the network.
    final cachedResults = await Future.wait<Object?>([
      repository.readCachedCharacters(page: 1, limit: 10),
      repository.readCachedPlanets(page: 1, limit: 10),
    ]);

    final cachedCharacterPage =
        cachedResults[0] as Page<Character>?;
    final cachedPlanetPage = cachedResults[1] as Page<Planet>?;

        cachedCharacterPage != null || cachedPlanetPage != null;

      // invalidate these background requests.

      final planetSequence = ++_planetRequestSequence;

      Future<void>.delayed(Duration.zero, () async {
        await Future.wait<void>([

              sequence: characterSequence,
              replace: true,
              request: () => repository.fetchCharacters(
                page: 1,
                limit: 10,
              ),

            _runPlanetRequest(
              sequence: planetSequence,
              replace: true,
              request: () => repository.fetchPlanets(
                page: 1,
                limit: 10,
              ),
            ),
        ]);

        characterCurrentPage: cachedCharacterPage?.currentPage ?? 1,
        hasNextCharacters: cachedCharacterPage?.hasNext ?? false,

        isLoadingPlanets: true,
        planetError: null,

      repository.fetchPlanets(page: 1, limit: 10),

    final planetResult =

class DragonBallFavoritesController extends Notifier<Set<int>> {
  static const _storageKey = 'dragonball_favorite_character_ids';

  bool _loaded = false;
  bool _restoreStarted = false;

  // IDs toggled while saved favorites are being restored.
  // Each ID's presence represents an odd number of toggles.
  final Set<int> _pendingToggles = {};

  // Serializes persistence writes so an older snapshot cannot
  // overwrite a newer snapshot after completing out of order.
  Future<void> _writeQueue = Future<void>.value();

  @override
  Set<int> build() {
    _startRestore();
    return const <int>{};
  }

  void _startRestore() {
    if (_loaded || _restoreStarted) return;

    _restoreStarted = true;
    _restoreFavorites();
  }

  void replaceAll(Set<int> characterIds) {
    final validIds = characterIds.where((id) => id > 0).toSet();

    _loaded = true;
    _restoreStarted = false;
    _pendingToggles.clear();

    state = Set<int>.unmodifiable(validIds);
    _enqueueSave();
  }

  Future<void> _restoreFavorites() async {
    try {
      final prefs = await SharedPreferences.getInstance();
      if (_loaded) return;

      final savedIds = (prefs.getStringList(_storageKey) ?? const <String>[])
          .map(int.tryParse)
          .whereType<int>()
          .toSet();

      final hadPendingToggles = _pendingToggles.isNotEmpty;

      // Merge taps that occurred while storage was loading.
      // Toggle semantics mean each pending ID must be inverted once.
      for (final id in _pendingToggles) {
        if (!savedIds.add(id)) {
          savedIds.remove(id);
        }
      }

      _pendingToggles.clear();

      state = Set<int>.unmodifiable(savedIds);
      _loaded = true;
      _restoreStarted = false;

      // No write is needed if restoration did not change anything.
      // If the user toggled during restoration, persist the merged set.
      if (hadPendingToggles) {
        _enqueueSave();
      }
    } catch (error) {
      _restoreStarted = false;

      // Keep pending toggles so a later toggle can retry restoration.
      debugPrint('Failed to restore Dragon Ball favorites: $error');
    }
  }

  void toggle(int characterId) {
    final next = Set<int>.of(state);

    if (!next.add(characterId)) {
      next.remove(characterId);
    }

    // Update the UI immediately; persistence happens asynchronously.
    state = Set<int>.unmodifiable(next);

    if (!_loaded) {
      if (!_pendingToggles.add(characterId)) {
        _pendingToggles.remove(characterId);
      }

      // Retry restoration if a previous attempt failed.
      _startRestore();
      return;
    }

    _enqueueSave();
  }

  void _enqueueSave() {
    // Capture this state snapshot now. The queue guarantees snapshots
    // are written in the same order in which they were enqueued.
    final ids = state.toList();

    _writeQueue = _writeQueue.then((_) async {
      try {
        final prefs = await SharedPreferences.getInstance();

        await prefs.setStringList(
          _storageKey,
          ids.map((id) => id.toString()).toList(),
        );
      } catch (error) {
        debugPrint('Failed to save Dragon Ball favorites: $error');
      }
    });
  }

  bool contains(int characterId) => state.contains(characterId);
}

// LEARN (Riverpod, the big idea of this file): ONE source of truth, then
// small DERIVED providers on top of it.
//
//   allCharactersProvider      (network/cache, all 58 characters, loaded once)
//        |
//        +--> exploreGroupsProvider   (race/affiliation chips + counts)
//        +--> exploreVisibleProvider  (filtered + searched + sorted list)
//             ^
//   exploreSelectionProvider (mode, group, query, sort chosen by the user)
//
// Groups come from the REAL data, so a new race added to the API appears as
// a chip automatically. A hardcoded list can never get out of sync.

enum ExploreMode { all, race, affiliation }

extension ExploreModeLabel on ExploreMode {
  String label(AppStrings s) => switch (this) {
        ExploreMode.all => s.modeAll,
        ExploreMode.race => s.modeRace,
        ExploreMode.affiliation => s.modeAffiliation,

enum ExploreSort { nameAsc, nameDesc, kiDesc }

// ---------------------------------------------------------------------------
// 1) Load every character (all pages), with offline fallback
// ---------------------------------------------------------------------------

final allCharactersProvider = FutureProvider<List<Character>>((ref) async {
  final repo = ref.watch(dragonBallRepositoryProvider);

  // A Map keyed by id removes duplicates if pages overlap.
  final byId = <int, Character>{};
  Failure? lastFailure;

  // The upper bound (50 pages) protects against an infinite loop if the API
  // ever keeps answering hasNext = true.
  for (var pageNumber = 1; pageNumber <= 50; pageNumber++) {
    final result = await repo.fetchCharacters(page: pageNumber, limit: 10);

    Page<Character>? page;
    switch (result) {
      case Ok(:final value):
        page = value;
      case Err(:final failure):
        lastFailure = failure;
    }

    // Offline fallback: the repository already wrote every successful page
    // to the JSON cache, so a failed request can be served from disk.
    page ??= await repo.readCachedCharacters(page: pageNumber, limit: 10);

    if (page == null) {
      if (byId.isEmpty) {
        throw DragonBallRequestException(
          lastFailure ?? const UnknownFailure('No data is available.'),
        );
      }
      break; // keep the pages we already have
    }

    for (final character in page.items) {
      byId[character.id] = character;
    }
    if (!page.hasNext) break;
  }

  return byId.values.toList(growable: false);
});

@immutable
class ExploreSelection {
  const ExploreSelection({
    this.mode = ExploreMode.all,
    this.group, // null = "all races" / "all affiliations"
    this.query = '',
    this.sort = ExploreSort.nameAsc,
  });

  final ExploreMode mode;
  final String? group;
  final String query;
  final ExploreSort sort;

  // `group` is nullable on purpose, so "keep the old value" and "set to
  // null" need different signals: that is what clearGroup is for.
  ExploreSelection copyWith({
    ExploreMode? mode,
    String? group,
    bool clearGroup = false,
    String? query,
    ExploreSort? sort,
  }) {
    return ExploreSelection(
      mode: mode ?? this.mode,
      group: clearGroup ? null : (group ?? this.group),
      query: query ?? this.query,
      sort: sort ?? this.sort,
    );
  }
}

final exploreSelectionProvider =
    NotifierProvider<ExploreSelectionController, ExploreSelection>(
  ExploreSelectionController.new,

class ExploreSelectionController extends Notifier<ExploreSelection> {
  @override
  ExploreSelection build() => const ExploreSelection();

  /// Changing the mode resets the group (a "Saiyan" chip means nothing
  /// under Affiliation) but keeps the search text and the sort.
  void setMode(ExploreMode mode) {
    state = ExploreSelection(mode: mode, query: state.query, sort: state.sort);
  }

  void setGroup(String? group) {
    state = state.copyWith(group: group, clearGroup: group == null);
  }

  void setQuery(String query) => state = state.copyWith(query: query);

  void setSort(ExploreSort sort) => state = state.copyWith(sort: sort);

  /// Used by breadcrumb taps: show exactly this group, search cleared.
  void jumpTo({required ExploreMode mode, String? group}) {
    state = ExploreSelection(mode: mode, group: group, sort: state.sort);
  }

  void reset() => state = ExploreSelection(sort: state.sort);
}

// 3) Derived data
// ---------------------------------------------------------------------------

typedef GroupCount = ({String label, int count}); // a Dart 3 record

/// Chips for the current mode: label + how many characters.
final exploreGroupsProvider = Provider<List<GroupCount>>((ref) {
  final all = ref.watch(allCharactersProvider).asData?.value ??
      const <Character>[];
  final mode = ref.watch(exploreSelectionProvider.select((s) => s.mode));

  if (mode == ExploreMode.all) return const [];

  final counts = <String, int>{};
  for (final c in all) {
    final key = mode == ExploreMode.race ? c.race : c.affiliation;
    counts[key] = (counts[key] ?? 0) + 1;
  }

  final groups = [
    for (final e in counts.entries) (label: e.key, count: e.value),
  ]..sort((a, b) {
      final byCount = b.count.compareTo(a.count);
      return byCount != 0 ? byCount : a.label.compareTo(b.label);
    });

  return groups;
});

/// The list shown on screen. Riverpod caches the result: sorting runs only
/// when the data or the selection changes, NOT on every widget rebuild
/// (the old home screen sorted inside build()).
final exploreVisibleProvider = Provider<List<Character>>((ref) {
  final all = ref.watch(allCharactersProvider).asData?.value ??
      const <Character>[];
  final sel = ref.watch(exploreSelectionProvider);
  final query = sel.query.trim().toLowerCase();

  final list = all.where((c) {
    final inGroup = switch (sel.mode) {
      ExploreMode.all => true,
      ExploreMode.race => sel.group == null || c.race == sel.group,
      ExploreMode.affiliation =>
        sel.group == null || c.affiliation == sel.group,
    };
    return inGroup && (query.isEmpty || c.name.toLowerCase().contains(query));
  }).toList();

  switch (sel.sort) {
    case ExploreSort.nameAsc:
      list.sort((a, b) => a.name.toLowerCase().compareTo(b.name.toLowerCase()));
    case ExploreSort.nameDesc:
      list.sort((a, b) => b.name.toLowerCase().compareTo(a.name.toLowerCase()));
    case ExploreSort.kiDesc:
      // Parse each ki string ONCE, not on every comparison.
      final power = {for (final c in list) c.id: powerOf(c.ki, c.maxKi)};
      list.sort((a, b) => power[b.id]!.compareTo(power[a.id]!));
  }

  return list;
});

class _FullScreenImageViewerState
    extends State<FullScreenImageViewer> {
  late final PageController _pageController;
  late int _currentPageIndex;
  late int _currentIndex;

  int get _initialIndex =>
      widget.initialIndex.clamp(0, widget.images.length - 1);

  bool get _hasMultipleImages => widget.images.length > 1;

  @override
  void initState() {
    super.initState();

    _currentIndex = _initialIndex;

    final initialPage = widget.pageNotifier?.value ??
        (_hasMultipleImages
            ? widget.images.length * 1000 + _currentIndex
            : 0);

    _currentPageIndex = initialPage;
    _pageController = PageController(initialPage: initialPage);

    widget.pageNotifier?.addListener(_syncToSharedPage);
  }

  Future<void> _closeViewer() async {
    final target = widget.pageNotifier?.value;

    if (target != null) {
      widget.pageNotifier!.value = target;
    }

    widget.onBeforePop?.call();

    if (mounted) {
      Navigator.of(context).pop();
    }
  }

  void _syncToSharedPage() {
    if (!mounted || !_pageController.hasClients) return;

    final target = widget.pageNotifier?.value;
    if (target == null) return;

    final current = _pageController.page;

    if (current != null && (current - target).abs() < 0.01) {
      return;
    }

    _pageController.animateToPage(
      target,
      duration: const Duration(milliseconds: 750),
      curve: Curves.easeInOutCubicEmphasized,
    );
  }

  @override
  void dispose() {
    widget.pageNotifier?.removeListener(_syncToSharedPage);
    _pageController.dispose();
    super.dispose();
  }

      height: double.infinity,
      placeholder: const Center(
        child: CircularProgressIndicator(
          color: Colors.white,
        ),
      ),
      showLoadingIndicator: true,
      errorWidget: const Icon(
        Icons.broken_image_outlined,
        color: Colors.white70,
        size: 64,
      ),
    );

    // Only the currently visible page should own this Hero tag.
    if (pageIndex == _currentPageIndex) {
      return Center(
        child: Hero(
          tag: image.heroTag,
          child: imageWidget,
        ),
      );
    }

    return Center(child: imageWidget);
  }

//   onCrumbTap() -> what happens when a crumb is tapped (side effects)

/// One tappable piece of the path, e.g. "Frieza Race".
@immutable
class Crumb {
  const Crumb(this.label, {this.onTap});

  final String label;
  final VoidCallback? onTap;
}

sealed class PathSource {
  const PathSource();

  /// Crumbs BEFORE the character name. Empty list = just the character name.
  List<String> labels(AppStrings s);

  /// Default: go back to the screen that opened this character.
  void onCrumbTap(int index, WidgetRef ref, BuildContext context) {
    Navigator.of(context).pop();
  }
}

/// Opened without any group context (All, or the old Characters tab).
final class FromAll extends PathSource {
  const FromAll();

  @override
  List<String> labels(AppStrings s) => const [];
}

/// Opened from the Explore tab. Path examples:
///   mode all                 -> Zarbon
///   race, group Frieza Race  -> Frieza Race › Zarbon
///   race, no chip selected   -> Race › Zarbon
///   with search "za"         -> Frieza Race › “za” › Zarbon
final class FromExplore extends PathSource {
  const FromExplore({required this.mode, this.group, this.query = ''});

  final ExploreMode mode;
  final String? group;
  final String query;

  @override
  List<String> labels(AppStrings s) => [
        if (mode != ExploreMode.all) group ?? mode.label(s),
        if (query.isNotEmpty) '“$query”',
      ];

  @override
  void onCrumbTap(int index, WidgetRef ref, BuildContext context) {
    final isGroupCrumb = mode != ExploreMode.all && index == 0;
    if (isGroupCrumb) {
      // Tapping "Frieza Race" shows that group again, without the search.
      ref.read(exploreSelectionProvider.notifier).jumpTo(mode: mode, group: group);
    }
    // Tapping the search crumb keeps the list exactly as it was.
    Navigator.of(context).pop();
  }
}

final class FromFavorites extends PathSource {
  const FromFavorites();

  @override
  List<String> labels(AppStrings s) => [s.favoritesCrumb];
}

final class FromPlanet extends PathSource {
  const FromPlanet(this.planetName);

  final String planetName;

  @override
  List<String> labels(AppStrings s) => [planetName];
}

/// Everything the detail screen needs: where we came from, which characters

    required this.index,

}

class SearchDebouncer {

      _timer = null;

  }

    _timer?.cancel();
    _timer = null;
  }

    cancel();
  }
}

Future<void> _importFavorites() async {
  try {
    final files = await FilePicker.pickFiles(
      type: FileType.custom,
      allowedExtensions: ['json'],
    );

    // Your installed API returns a List<PlatformFile>.
    if (files.isEmpty || !mounted) return;

    final selectedFile = files.first;
    final fileBytes = await selectedFile.readAsBytes();

    final dynamic decoded = jsonDecode(
      utf8.decode(fileBytes),
    );

    if (decoded is! Map ||
        decoded['schemaVersion'] != 1 ||
        decoded['favoriteCharacterIds'] is! List) {
      _message('This is not a supported favorites backup.');
      return;
    }

    final rawIds = decoded['favoriteCharacterIds'] as List;
    final ids = <int>{};

    for (final rawId in rawIds) {
      final int? id = rawId is int
          ? rawId
          : rawId is String
              ? int.tryParse(rawId)
              : null;

      if (id == null || id <= 0) {
        _message('The backup contains an invalid character ID.');
        return;
      }

      ids.add(id);
    }

    final confirmed = await _confirm(
      title: 'Import favorites?',
      message:
          'This will replace your current favorites with '
          '${ids.length} character IDs from the selected backup.',
      action: 'Import',
    );

    if (!confirmed || !mounted) return;

    ref
        .read(dragonBallFavoritesProvider.notifier)
        .replaceAll(ids);

    _message('Favorites imported successfully.');
  } on FormatException {
    _message('The selected file contains invalid JSON.');
  } catch (error) {
    _message('Could not import favorites: $error');
  }
}

class SplashScreen extends ConsumerStatefulWidget {
  const SplashScreen({super.key});

  @override
  ConsumerState<SplashScreen> createState() => _SplashScreenState();
}

class _SplashScreenState extends ConsumerState<SplashScreen> {
  bool _navigationScheduled = false;
  bool _preloadStarted = false;
  bool _imagesReady = false;

  @override
  Widget build(BuildContext context) {

    // Start loading the catalogues while the splash screen is visible.
    final catalog = ref.watch(dragonBallControllerProvider);
    final data = catalog.asData?.value;

    final isLoadingCharacters =
        data?.isLoadingCharacters ?? true;
    final isLoadingPlanets =
        data?.isLoadingPlanets ?? true;

    Future<void> _prepareInitialImages(
      List<Character> characters,
      List<Planet> planets,
    ) async {
      await preloadImageUrls(
        context,
        [
          ...characters.map((character) => character.imageUrl),
          ...planets.map((planet) => planet.imageUrl),
        ],
      );

      if (mounted) {
        setState(() => _imagesReady = true);
      }
    }    

    // Navigate only after both initial catalogue requests have finished.
    // Failed requests also finish loading, allowing the home screen
    // to display its existing error and retry UI.
    final isReady = data != null &&
        !isLoadingCharacters &&
        !isLoadingPlanets;

    if (isReady && !_preloadStarted) {
      _preloadStarted = true;

      WidgetsBinding.instance.addPostFrameCallback((_) {
        if (!mounted || data == null) return;

        _prepareInitialImages(data.characters, data.planets);
      });
    }

    if (isReady && _imagesReady && !_navigationScheduled) {
      _navigationScheduled = true;

      WidgetsBinding.instance.addPostFrameCallback((_) {
        if (!mounted) return;

        Navigator.of(context).pushReplacement(
          PageRouteBuilder<void>(
            pageBuilder: (_, animation, secondaryAnimation) =>
                const DragonBallHomeScreen(),
            transitionsBuilder: (_, animation, secondaryAnimation, child) {
              return FadeTransition(opacity: animation, child: child);
            },
            transitionDuration: const Duration(milliseconds: 350),
          ),
        );
      });
    }

    final theme = Theme.of(context);
    final colors = theme.colorScheme;

    final String status;
    if (data == null ||
        (isLoadingCharacters && isLoadingPlanets)) {
      status = 'Loading characters and planets...';
    } else if (isLoadingCharacters) {
      status = 'Loading characters...';
    } else if (isLoadingPlanets) {
      status = 'Loading planets...';
    } else {
      status = 'Preparing your universe...';
    }


    return Scaffold(
      backgroundColor: theme.scaffoldBackgroundColor,
      body: SafeArea(
        child: Center(
          child: Padding(
            padding: const EdgeInsets.symmetric(horizontal: 24),
            child: Column(
              mainAxisAlignment: MainAxisAlignment.center,
              children: [
                // Show the complete image without a circular crop.
                Image.asset(
                  'assets/dragonball-8.png',
                  width: 260,
                  height: 260,
                  fit: BoxFit.contain,
                  errorBuilder: (context, error, stackTrace) {
                    debugPrint('Splash image error: $error');

                    return const Text('Unable to load image');
                  },
                ),

                const SizedBox(height: 24),

                Text(
                  'Dragon Ball Explorer',
                  textAlign: TextAlign.center,
                  style: theme.textTheme.headlineSmall?.copyWith(
                    fontWeight: FontWeight.bold,
                  ),
                ),

                const SizedBox(height: 10),

                Text(
                  'Explore characters and planets',
                  textAlign: TextAlign.center,
                  style: theme.textTheme.bodyMedium?.copyWith(
                    color: colors.onSurfaceVariant,
                  ),
                ),

// LEARN (layout + semantics):
//  * Wrap lets the path flow onto a second line on narrow screens or with
//    large text, instead of overflowing like a Row would.
//  * The LAST crumb is the current page (bold, not tappable); earlier crumbs
//    are links that run Crumb.onTap.
//  * AnimatedSwitcher cross-fades the last crumb when you swipe to another
//    character, with no controller needed.
//  * One Semantics label reads the whole path to a screen reader.
class BreadcrumbBar extends StatelessWidget {
  const BreadcrumbBar({super.key, required this.crumbs, this.trailing});

  final List<Crumb> crumbs;

  /// e.g. "3 / 12" position text.
  final Widget? trailing;

  @override
  Widget build(BuildContext context) {
    final theme = Theme.of(context);
    final scheme = theme.colorScheme;
    final last = crumbs.length - 1;

    return Semantics(
      container: true,
      label: 'Path: ${crumbs.map((c) => c.label).join(', ')}',
      child: Material(
        color: scheme.surfaceContainerLow,
        child: Padding(
          padding: const EdgeInsets.symmetric(
            horizontal: AppSpace.x4,
            vertical: AppSpace.x2,
          ),
          child: Row(
            children: [
              Expanded(
                child: Wrap(
                  crossAxisAlignment: WrapCrossAlignment.center,
                  spacing: AppSpace.x1,
                  runSpacing: AppSpace.x1,
                  children: [
                    for (var i = 0; i < crumbs.length; i++) ...[
                      if (i > 0)
                        Icon(
                          AppIcons.caretRight,
                          size: 14,
                          color: scheme.onSurfaceVariant,
                        ),
                      if (i == last)
                        AnimatedSwitcher(
                          duration: AppMotion.base,
                          child: Padding(
                            key: ValueKey<String>(crumbs[i].label),
                            padding: const EdgeInsets.symmetric(
                              horizontal: 6,
                              vertical: 4,
                            ),
                            child: Text(
                              crumbs[i].label,
                              style: theme.textTheme.titleSmall,
                            ),
                          ),
                        )
                      else
                        InkWell(
                          borderRadius: BorderRadius.circular(AppRadius.sm),
                          onTap: crumbs[i].onTap,
                          child: Padding(
                            padding: const EdgeInsets.symmetric(
                              horizontal: 6,
                              vertical: 4,
                            ),
                            child: Text(
                              crumbs[i].label,
                              style: theme.textTheme.titleSmall?.copyWith(
                                color: scheme.primary,

                            ),
                          ),
                        ),
                    ],
                  ],
                ),
              ),
              if (trailing != null) trailing!,
            ],
          ),
        ),
      ),
    );
  }

import 'package:cached_network_image/cached_network_image.dart';
import 'package:flutter/foundation.dart';
import 'package:flutter/material.dart';

final Set<String> _reportedImageErrors = <String>{};

void _showImageError(
  BuildContext context,
  String imageUrl,
) {
  if (!_reportedImageErrors.add(imageUrl)) return;

  WidgetsBinding.instance.addPostFrameCallback((_) {
    if (!context.mounted) return;

    final messenger = ScaffoldMessenger.maybeOf(context);
    if (messenger == null) return;

    messenger
      ..hideCurrentSnackBar()
      ..showSnackBar(
        const SnackBar(
          content: Text(
            'Could not load an image. Check your connection and try again.',
          ),
          behavior: SnackBarBehavior.floating,
          duration: Duration(seconds: 3),
        ),
      );
  });
}

Widget buildCharacterImage(
  String imageUrl, {
  BoxFit fit = BoxFit.contain,
  double? width,
  double? height,
  Widget? placeholder,
  Widget? errorWidget,
  bool showLoadingIndicator = false,
}) {
  final loadingWidget = showLoadingIndicator
      ? (placeholder ??
          const Center(
            child: CircularProgressIndicator(strokeWidth: 2),
          ))
      : const SizedBox.shrink();

  final fallbackWidget =
      errorWidget ?? const Icon(Icons.broken_image_outlined);

  if (imageUrl.trim().isEmpty) {
    return fallbackWidget;
  }

  if (kIsWeb) {
    return Image.network(
      imageUrl,
      fit: fit,
      width: width,
      height: height,
      gaplessPlayback: true,
      frameBuilder: (context, child, frame, wasSynchronouslyLoaded) {
        if (wasSynchronouslyLoaded || frame != null) {
          return AnimatedOpacity(
            opacity: 1,
            duration: const Duration(milliseconds: 150),
            curve: Curves.easeOut,
            child: child,
          );
        }

        return loadingWidget;
      },
      errorBuilder: (context, error, stackTrace) {
        _showImageError(context, imageUrl);
        return fallbackWidget;
      },
    );
  }

  return CachedNetworkImage(
    imageUrl: imageUrl,
    fit: fit,
    width: width,
    height: height,
    useOldImageOnUrlChange: true,
    fadeInDuration: const Duration(milliseconds: 150),
    fadeOutDuration: Duration.zero,
    placeholder: (context, url) => loadingWidget,
    errorWidget: (context, url, error) {
      _showImageError(context, url);
      return fallbackWidget;
    },
  );
}

import 'package:flutter/material.dart';

// LEARN (several small topics in one widget):
//  * ConsumerWidget + ref.watch(provider.select(...)): rebuilds ONLY when
//    this one character's favorite flag changes, not when any favorite does.
//  * HapticFeedback: a tiny vibration on tap (mobile only, ignored on web).
//  * TweenAnimationBuilder with a ValueKey: when the key changes, the tween
//    restarts, so the heart "pops" (scale 0.6 -> 1.0 with an elastic curve)
//    every time it is toggled.
class FavoriteHeart extends ConsumerWidget {
  const FavoriteHeart({
    super.key,
    required this.characterId,
    this.overlay = true,
  });

  final int characterId;

  /// true: round semi-transparent background (on top of images).
  final bool overlay;

  @override
  Widget build(BuildContext context, WidgetRef ref) {
    final s = ref.watch(stringsProvider);
    final scheme = Theme.of(context).colorScheme;

    final isFavorite = ref.watch(
      dragonBallFavoritesProvider.select((ids) => ids.contains(characterId)),
    );

    final noMotion = MediaQuery.disableAnimationsOf(context);

    return IconButton(
      tooltip: isFavorite ? s.removeFavorite : s.addFavorite,
      visualDensity: overlay ? VisualDensity.compact : VisualDensity.standard,
      style: overlay
          ? IconButton.styleFrom(
              backgroundColor: scheme.surface.withValues(alpha: 0.85),
              shape: const CircleBorder(),
            )
          : null,
      onPressed: () {
        HapticFeedback.lightImpact();
        ref.read(dragonBallFavoritesProvider.notifier).toggle(characterId);
      },
      icon: TweenAnimationBuilder<double>(
        key: ValueKey<bool>(isFavorite),
        tween: Tween<double>(begin: isFavorite ? 0.6 : 1.0, end: 1.0),
        duration: noMotion ? Duration.zero : AppMotion.slow,
        curve: Curves.elasticOut,
        builder: (context, scale, child) =>
            Transform.scale(scale: scale, child: child),
        child: Icon(
          isFavorite ? AppIcons.heartFill : AppIcons.heart,
          color: isFavorite ? Colors.redAccent : null,
        ),
      ),
    );
  }
}

import 'package:cached_network_image/cached_network_image.dart';
import 'package:flutter/foundation.dart';
import 'package:flutter/material.dart';

Future<void> preloadImageUrls(
  BuildContext context,
  Iterable<String> urls,
) async {
  final uniqueUrls = urls
      .map((url) => url.trim())
      .where((url) => url.isNotEmpty)
      .toSet();

  await Future.wait(
    uniqueUrls.map((url) async {
      final ImageProvider provider = kIsWeb
          ? NetworkImage(url)
          : CachedNetworkImageProvider(url);

      try {
        await precacheImage(provider, context);
      } catch (_) {
        // Loading failures are handled by buildCharacterImage.
      }
    }),
  );
}

import 'dart:math' as math;

import 'package:flutter/material.dart';

import '../core/design/app_tokens.dart';
import '../core/utils/ki_parser.dart';

// LEARN (implicit animation + accessibility):
//  * TweenAnimationBuilder animates a value from begin to end with no
//    AnimationController and no dispose(). The bar "fills in" when it appears.
//  * MediaQuery.disableAnimationsOf respects the user's reduce-motion setting
//    (system or the app's own toggle, which main.dart merges in).
//  * Semantics gives screen readers ONE sentence instead of two colored
//    rectangles they cannot describe.
//
// Two layers: faint bar = max ki, solid bar = current ki (both log scale).
class KiBar extends StatelessWidget {
  const KiBar({
    super.key,
    required this.ki,
    required this.maxKi,
    this.height = 6,
    this.showLabels = false,
    this.kiLabel = 'Ki',
    this.maxLabel = 'Max',
  });

  final String ki;
  final String maxKi;
  final double height;
  final bool showLabels;
  final String kiLabel;
  final String maxLabel;

  @override
  Widget build(BuildContext context) {
    final scheme = Theme.of(context).colorScheme;
    final theme = Theme.of(context);

    final kiValue = parseKi(ki);
    final maxValue = parseKi(maxKi);

    final kiFraction = kiScale(kiValue);
    // Max can never look smaller than current (bad API data happens).
    final maxFraction = math.max(kiScale(maxValue), kiFraction);

    final duration =
        MediaQuery.disableAnimationsOf(context) ? Duration.zero : AppMotion.slow;

    final bar = TweenAnimationBuilder<double>(
      tween: Tween<double>(begin: 0, end: 1),
      duration: duration,
      curve: AppMotion.curve,
      builder: (context, t, _) {
        return ClipRRect(
          borderRadius: BorderRadius.circular(height),
          child: SizedBox(
            height: height,
            child: Stack(
              children: [
                Positioned.fill(
                  child: ColoredBox(
                    color: scheme.outlineVariant.withValues(alpha: 0.5),
                  ),
                ),
                FractionallySizedBox(
                  alignment: Alignment.centerLeft,
                  widthFactor: maxFraction * t,
                  child: ColoredBox(
                    color: scheme.secondary.withValues(alpha: 0.45),
                    child: const SizedBox.expand(),
                  ),
                ),
                FractionallySizedBox(
                  alignment: Alignment.centerLeft,
                  widthFactor: kiFraction * t,
                  child: ColoredBox(
                    color: scheme.primary,
                    child: const SizedBox.expand(),
                  ),
                ),
              ],
            ),
          ),
        );
      },
    );

    return Semantics(
      label: '$kiLabel ${formatKi(kiValue)}, $maxLabel ${formatKi(maxValue)}',
      child: ExcludeSemantics(
        child: showLabels
            ? Column(
                crossAxisAlignment: CrossAxisAlignment.stretch,
                children: [
                  bar,
                  const SizedBox(height: AppSpace.x2),
                  Row(
                    mainAxisAlignment: MainAxisAlignment.spaceBetween,
                    children: [
                      Text(
                        '$kiLabel ${formatKi(kiValue)}',
                        style: theme.textTheme.labelMedium,
                      ),
                      Text(
                        '$maxLabel ${formatKi(maxValue)}',
                        style: theme.textTheme.labelMedium?.copyWith(
                          color: scheme.secondary,
                        ),
                      ),
                    ],
                  ),
                ],
              )
            : bar,
      ),
    );
  }
}

/// Tinted pill with an icon and a label (race, affiliation, tier, status).
class InfoChip extends StatelessWidget {
  const InfoChip({
    super.key,
    required this.label,
    required this.icon,
    required this.color,
    this.dense = false,
    this.onTap,
  });

  final String label;
  final IconData icon;
  final Color color;
  final bool dense;
  final VoidCallback? onTap;

  @override
  Widget build(BuildContext context) {
    final scheme = Theme.of(context).colorScheme;

    // Pure orange text on a light background has weak contrast. Mixing the
    // category color with the normal text color keeps the hue but is readable.
    final textColor = Color.lerp(color, scheme.onSurface, 0.35)!;

    return Material(
      color: color.withValues(alpha: 0.14),
      shape: const StadiumBorder(),
      child: InkWell(
        customBorder: const StadiumBorder(),
        onTap: onTap,
        child: Padding(
          padding: EdgeInsets.symmetric(
            horizontal: dense ? 8 : 10,
            vertical: dense ? 3 : 5,
          ),
          child: Row(
            mainAxisSize: MainAxisSize.min,
            children: [
              Icon(icon, size: dense ? 12 : 14, color: textColor),
              const SizedBox(width: 4),
              Flexible(
                child: Text(
                  label,
                  maxLines: 1,
                  overflow: TextOverflow.ellipsis,
                  style: Theme.of(context).textTheme.labelSmall?.copyWith(
                        color: textColor,
                        fontWeight: FontWeight.w700,
                      ),
                ),
              ),
            ],
          ),
        ),
      ),
    );
  }
}
